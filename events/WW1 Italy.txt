#Italy Win (Send Option to turkey to surrender or not)
country_event = {
	id = ww1_italy.2
	title = ww1_italy.2.t
	desc = ww1_italy_desc.2
	picture = GFX_report_event_generic_meeting

	is_triggered_only = yes

	timeout_days = 4

	option = {
		name = ww1_italy.2.a
		
		TUR = {
			country_event = { id = ww1_ottoman.1 days = 5 }
	    }
		
		ai_chance = {
			factor = 100
		}

		ITA = {
			complete_national_focus = ITA_focus_Triumph_in_Tripoli
			if = {
				limit = {
					NOT = {
						OR = {
							has_completed_focus = ITA_focus_Hire_a_Chief_of_Army
							has_completed_focus = ITA_focus_Hire_Douhet
							has_completed_focus = ITA_focus_Hire_Pellegrini
							has_completed_focus = ITA_focus_Appoint_Chief_of_Navy
						}
					}
				}
				complete_national_focus = ITA_focus_Hire_a_Chief_of_Army
			}
# intent: if none of the four has been done, give the player Cadorna to not break anything and then complete Armed Forces Review
			complete_national_focus = ITA_focus_armed_forces_review
		}
	}
}
