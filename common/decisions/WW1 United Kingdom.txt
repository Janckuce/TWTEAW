			}
		}

		selectable_mission = no
		
		days_mission_timeout = 90

		}

		}
		timeout_effect = {
			AST = {
				random_core_state = {
					limit = {
						NOT = { has_state_flag = ENG_development_marker }
						is_fully_controlled_by = AST
						
						free_building_slots = {
							building = arms_factory
							size > 0
							include_locked = yes
						}
						free_building_slots = {
							building = industrial_complex
							size > 1
							include_locked = yes
						}
					}
					
					set_state_flag = ENG_development_marker # Marks the selected state
					custom_effect_tooltip = ENG_ast_dev_tt5
				}
				
			}
			hidden_effect = {
				add_to_variable = { AST_investments = 1 }
				if = {
					limit = {
						check_variable = {
							var = AST_investments
							value = AST.num_core_states
							compare = greater_than_or_equals
						}
					}
					add_to_variable = { ENG_industrial_consumer_goods_expected_value = -0.01 tooltip = consumer_goods_expected_value_tt }
				}
				else = {
					activate_mission = ENG_invest_ast
				}
			}
		}
	}
