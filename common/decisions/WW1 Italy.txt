ITA_Lybia_War_Category = {
	ITA_Lybia_War_Timer = {
		days_mission_timeout = 250
		is_good = no
		allowed = {
			tag = ITA
		}

		activation = {
			AND = {
				has_war_with = TUR
				has_country_flag = {
					flag = ITA_attack_turkey
				}
				NOT = {
					TUR = {
						has_country_flag = TUR_surrendered_lybia_war
					}
					has_country_flag = ITA_surrendered_lybia_war
				}
			}
		}

		available = {
			has_country_flag = {
				flag = ITA_attack_turkey
				days > 250
			}
			controls_state = 448
			controls_state = 450
			controls_state = 451
		}
		cancel_trigger = {
			OR = {
				has_country_flag = ITA_Lose_Libya
				has_country_flag = ITA_Win_Libya
				has_country_flag = ITA_surrendered_lybia_war
				TUR = {
					has_country_flag = TUR_surrendered_lybia_war
				}
			}
		}

		timeout_effect = {

    	}
	}

	ITA_lybia_war_monthly_rewards = {
		days_mission_timeout = 20
		is_good = yes
		allowed = {
			tag = ITA
		}

		activation = {
			AND = {
				has_war_with = TUR
				has_country_flag = {
					flag = ITA_attack_turkey
				}
				NOT = {
					TUR = {
						has_country_flag = TUR_surrendered_lybia_war
					}
					has_country_flag = ITA_surrendered_lybia_war
				}
			}
		}

		available = {
			always = no
		}

		highlight_states = {
			highlight_state_targets = {
				state = 449
			}
			highlight_provinces = {
				1204
			}
			highlight_color_while_active = 1
			highlight_color_before_active = 1
		}

		timeout_effect = {
			if = {
				limit = {
					date < 1912.1.1
				}
				activate_mission = ITA_lybia_war_monthly_rewards
			}

			custom_effect_tooltip = lybia_war_wealth_extraction_tooltip

			if = {
				limit = {
					controls_province = 1204
				}
				add_offsite_building = { type = industrial_complex level = 1 }
			}
			if = {
				limit = {
					controls_state = 449
				}
				add_offsite_building = { type = industrial_complex level = 1 }
			} 
		}
	}

	ITA_Lybia_War_Surrender = {
		allowed = {
			original_tag = ITA
		}
		visible = {
			has_country_flag = ITA_attack_turkey
			NOT = {
				TUR = {
					has_country_flag = TUR_surrendered_lybia_war
				}
				has_country_flag = ITA_Lose_Libya
				has_country_flag = ITA_Win_Libya
			}
		}
		available = {
			has_country_flag = {
				flag = ITA_attack_turkey
				days > 100
			}
		}
		complete_effect = {
			set_country_flag = ITA_surrendered_lybia_war
		}
		fire_only_once = yes
		
	}
}
