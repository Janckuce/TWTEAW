on_actions = {
	
	#ROOT is new controller #FROM is old controller #FROM.FROM is state ID

	on_state_control_changed = {
		
		##UK Reconquering a state lost
		effect = {
			if = {
				limit = {
					FROM = { 						
						OR = {
							tag = RUS
							is_in_faction_with = RUS
						}  
					}
					ROOT = { 
						OR = { 
							is_in_faction_with = GER 
							tag = GER
						} 
					}
					FROM.FROM = {
						has_state_flag = RUS_controls_russian
					}
				}
				FROM.FROM = {  
				 	set_state_flag = GER_controls_russian
				 	clr_state_flag = RUS_controls_russian
				}

				RUS = { 
					RUS_bop_count_level_up = yes
				}
			}
		}
		effect = {
			if = {
				limit = {
					FROM = { 
					    OR = { 
							is_in_faction_with = GER 
							tag = GER
						}
					}
					ROOT = {
						OR = {
							tag = RUS
							is_in_faction_with = RUS
						} 
					}
					FROM.FROM = {
						has_state_flag = GER_controls_russian
					}
				}
				FROM.FROM = { 
				 	set_state_flag = RUS_controls_russian
				 	clr_state_flag = GER_controls_russian
				}

				RUS = {
					RUS_bop_count_level_down = yes
				}
			}
		}
	}
}
