#Important to put HIGHEST first
turkey_debuff_level_up = {
	
	if = {
		limit = { has_idea = ottomans_debuff4 }
		swap_ideas = {
			remove_idea = ottomans_debuff4
			add_idea = ottomans_debuff5
		}		
	}
	else_if = {
		limit = { has_idea = ottomans_debuff3 }
		swap_ideas = {
			remove_idea = ottomans_debuff3
			add_idea = ottomans_debuff4
		}		
	}
	else_if = {
		limit = { has_idea = ottomans_debuff2 }
		swap_ideas = {
			remove_idea = ottomans_debuff2
			add_idea = ottomans_debuff3
		}
	}
	else_if = {
		limit = { has_idea = ottomans_debuff1 }
		swap_ideas = {
			remove_idea = ottomans_debuff1
			add_idea = ottomans_debuff2
		}
	}
	else_if = {
		limit = { has_idea = ottomans_debuff0 }
		swap_ideas = {
			remove_idea = ottomans_debuff0
			add_idea = ottomans_debuff1
		}
	}
}
